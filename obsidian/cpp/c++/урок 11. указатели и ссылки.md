
# ссылки

ссылка - примерно то же самое что и переменная, разница в том, что ссылка не обращается к тому значению, которое было записано в этой переменной, а обращается лишь к адресу, где записана сама переменная.

```cpp
#include <iostream>
#include <string>

using namespace std;

int main() {
	setlocale(LC_ALL, "RU");
	
	/*
	int num = 10;
	int a = num;     занимают 8 байтов суммарно в памяти пк 
	
	каждый раз когда некая переменная должна хранить значение другой переменной, то не логично прописывать код таким образом.
	лучше использовать ссылки.
	*/
	
	//чтобы создрать из переменной ссылку , нужно прописать & :
	int num = 10;
	int &a = num;  //int &a теперь ссылка
	
	cout << &num << endl;
	cout << &a << endl;  //обращаемся к названию ячейки, где храниться переменная. адреса будут совпадать
	
	
	
	return 0;
}
```
это дает нам то, что теперь переменная занимает мало оперативки и хранит адрес, где записана переменная.
	и когда мы имеем адрес, где у нас записана предыдущая некая переменная, то мы можем по этому адресу обратиться и тут же пк выдаст нам то значение, которое записано по этому адресу в этой ячейке
		теперь мы храним просто адрес, а адрес занимает меньше оперативки

ссылка не содержит само по себе значение, они содержит лишь название той ячейки, где находится переменная.

```cpp
#include <iostream>

using namespace std;

int main() {
	setlocale(LC_ALL, "RU");
	
	int num = 10;
	int &a = num;
	
	a = 15;  //если поменять значение ссылки, то оно поменяется и для переменной
	cout << &num << " - " << num << endl;
	cout << &a << " - " << a << endl;
	//вывод. для ссылки и переменной будет одно и то же значение
	
	return 0;
}

```

т.е. ссылки это те конструкции, которые позволяют хранить программный адрес некой переменной, массива или объекта. ссылки удобны тем, что они всегда меньше нежели переменная, массив или объект


# указатели

ссылки хранят программный адрес
указатели указывают на программный адрес
- указатель можно создать и ничего ему не присваивать в отличие от ссылки
- можно очистить указатель
- можно делать сдвиг указателя

```cpp
#include <iostream>

using namespace std;

int main() {
    setlocale(LC_ALL, "RU");
    
    int val = 12;
    int* ptr1val;  //создание указателя (* или & можно ставить сразу после типа данных либо перед переменной)
    int* ptr2val = &val;  //надо указывать программное название ячейки
    // получился указатель int* ptrval который указывает на программную ячейку где записана переменная val.
    
    cout << &val << " + " << val << endl;  //val программный адрес + само значение val
    cout << ptr2val << " + " << *ptr2val << endl;  //указатель ptr2val и ячейка, на которую он указывает + и значение
    
    return 0;
}

```


поменять значение  переменной:

```cpp
#include <iostream>

using namespace std;

int main() {
	setlocale(LC_ALL, "RU");

	int val = 12;
	int* ptrval = &val;
	
	*ptrval = 20;  //поменять значение переменной (обязательно указывать * )
	ptrval = nullptr;  //удалить указатель (ни на что не указывает)
	
	cout << &val << " + " << val << endl;
	cout << ptrval << " + " << *ptrval << endl;
	
	return 0;
}
```

# практический пример

программа, в которую передается массив, и в функции находить мин. элемент среди всех элементов массива

```cpp
#include <iostream>

using namespace std;

void minimal(int* arr, int len) {
	int min = *arr;  //min = 1 элемент массива
	for (int i = 0; i < len; i++) {
		if(min > *(arr + i))  //смещение ячейки, т.е. к указателю будет добавляться единица (след элмемент массива)
			min = *(arr + i);  
	}
	cout << "min: " << min << endl;
}

int main() {
	setlocale(LC_ALL, "RU");
	
	int arr[] = {5, 7, 3, -2, 5};  //создрание массива
	minimal(arr, 5);  //передаем элементы массива в функцию

}
```
