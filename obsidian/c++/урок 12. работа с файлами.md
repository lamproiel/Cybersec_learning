для работы с файлами нужно подключить директиву `#include <fstream>`
	при помощи этого модуля можно записывать какие-то данные в файл, так и можем считывать данные из файла.

чтобы записывать некую информацию внутрь файла необходимо создать переменную, на основе класса `ofstream` 
	при помощи этого типа данных мы можем создать некий объект, за счет которого мы откроем файл, и за счет которого мы можем записать некую информацию внутрь открытого файла.

# создание, открытие файла и запись информации в файл

```cpp
#include <iostream>
#include <fstream>

using namespace std;

int main() {
    setlocale(LC_ALL, "RU");

//объект может называться как угодно
    ofstream file("text.txt", ios_base::out);  //параметры: в "" название файла, и формат открытия файла
    if(file.is_open()) {  //проверка, открыт ли файл. обращаемся к объекту и через . обращаемся к функции is_open() - она возвращает true/false. если вернула true, значит файл открыт и мы можем с ним работать, если false  то файл не открыт и мы не можем с ним работать.
	    file << "Hello World";  //внутрь открытого файла будет записан текст
	    file.close();  //функция для закрытия файла.
    }
    
    return 0;
}
```

	if (file.is_open()) {
	
		// string temp;  //в эту переменную будет записывать текст, полученный из файла
		// file >> temp;  //из файла считываем данные и записываем эти данные в переменную temp (ВЫВОД ТОЛЬКО 1 СЛОВА ИЗ ФАЙЛА)
		
		char temp[100];  //массив строк 
		file.getline(temp, 100)  //getline - функция, которая позволяет считать определенное кол-во символом из файла, 100 символом, и нужно передавать не строку string temp; , а массив строк.
		cout << temp << endl;  //вывод информации из переменной
		file.close();  //закрытие файла
	}
	
	return 0;
}
